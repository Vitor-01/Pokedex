async function limitarConcorrencia(tarefas, limite) {

    const resultados = [];
    let proximaTarefa = 0;

    async function trabalhador() {

        while (proximaTarefa < tarefas.length) {

            const indice = proximaTarefa;
            proximaTarefa++;

            resultados[indice] = await tarefas[indice]();
        }
    }

    const trabalhadores = [];

    for (let i = 0; i < limite; i++) {
        trabalhadores.push(trabalhador());
    }

    await Promise.all(trabalhadores);

    return resultados;
}
